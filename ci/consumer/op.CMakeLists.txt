# The CMakeLists.txt of a user's Op program, as the tutorial "Count a new hypergraph pattern (an
# Op)" quotes it (section 5) and ci/cmake_consumer.sh builds it, unchanged, against the installed
# package with ci/consumer/op.cu as my_patterns.cu. The script fails when the version below is not
# VERSION's major.minor.

# [tutorial: op consumer]
cmake_minimum_required(VERSION 3.30)
project(my_patterns CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
find_package(dyng 0.4 REQUIRED)
add_executable(my_patterns my_patterns.cu)
set_target_properties(my_patterns PROPERTIES CUDA_ARCHITECTURES native)
target_link_libraries(my_patterns PRIVATE dyng::dyng)
# [tutorial: op consumer end]
